<script is:inline>
  (function() {
    var p = window.location.pathname;
    if (!p.startsWith('/docs')) {
      window.location.replace('/docs' + (p.startsWith('/') ? p : '/' + p) + window.location.search + window.location.hash);
    }
  })();
</script>
